Help ~{
    FHDQuestReturn=Проложить маршрут к цели предложенного задания и оценить время пути
    FHDQuestRoute=Оценка по текущему оборудованию до посадки. Для дозаправок нужны деньги. Бои и ожидания не учтены.
    FHDQuestDeadline=Количество дней на выполнение текущего предложения задания, с учётом выбранных условий
    FHDQuestClear=Сбросить маршрут и метку задания с галактической карты
}
Script ^{
    QuestMarker ^{
        Mark ^{
            00=if(!star)exit;
            01=int markInfo=ShipFindCustomShipInfoByType(Player(),'FHDQuestMarkerState');
            02=if(markInfo<0)markInfo=ShipAddCustomShipInfo(Player(),'FHDQuestMarkerState','NoShow',0,0,0,'','','');
            03=ShipCustomShipInfoData(Player(),markInfo,1,Id(star));
            04=if(CurrentForm()=='Galaxy')ExecuteCodeFromString(GenerateCodeStringFromBlock('Script.QuestMarker.ApplyMark'),'star',star);
        }
        ApplyMark ^{
            00=if(!star || CurrentForm()!='Galaxy')exit;
            01=float galaxyWidth=CT('Constellations.GalaxySizeX');
            02=float galaxyHeight=CT('Constellations.GalaxySizeY');
            03=if(galaxyWidth<=0 || galaxyHeight<=0)exit;
            04=float x=CoordX(star);
            05=x=x/galaxyWidth;
            06=x=round(x*1151);
            07=float y=CoordY(star);
            08=y=y/galaxyHeight;
            09=y=round(y*787);
            10=InterfaceState('Galaxy','FHDQuestMarkV035',-1);InterfacePos('Galaxy','FHDQuestMarkV035',0,0,0);
            11=InterfacePos('Galaxy','FHDQuestMarkV035',-25+x,-25+y,0);
            12=InterfaceState('Galaxy','FHDQuestMarkV035',1);
        }
        ApplyPending ^{
            00=if(CurrentForm()!='Galaxy')exit;
            01=int markInfo=ShipFindCustomShipInfoByType(Player(),'FHDQuestMarkerState');
            02=if(markInfo<0)exit;
            03=int targetId=ShipCustomShipInfoData(Player(),markInfo,1);
            04=if(targetId<=0)exit;
            05=for(int i=0;i<GalaxyStars();i=i+1){dword candidate=GalaxyStar(i);if(candidate && Id(candidate)==targetId){ExecuteCodeFromString(GenerateCodeStringFromBlock('Script.QuestMarker.ApplyMark'),'star',candidate);break;}}
        }
        Place ^{
            00=if(!star)exit;unknown clearRoute=ImportedFunction('FHDQuestNative','FHD_RouteClear');clearRoute();
            01=ExecuteCodeFromString(GenerateCodeStringFromBlock('Script.QuestMarker.Mark'),'star',star);
            02=if(CurrentForm()!='Galaxy')RunChildForm('Galaxy',2);
        }
        Error ^{
            00=str problem='Не удалось построить маршрут.';
            01=if(status==-1)problem='Версия Rangers.exe не поддерживается или обработчик карты уже изменён другим модом.';
            02=if(status==-2)problem='Нет подходящего двигателя, бака или скорости для перелёта.';
            03=if(status==-3)problem='До цели нет маршрута с текущим оборудованием и доступными дозаправками.';
            04=if(status==-4)problem='Цель задания сейчас недоступна. Запросите задание снова.';
            05=if(status==-5)problem='Система цели ещё не открыта на галакарте.';
            06=if(status==-6)problem='Маршрут превышает предел расчёта (4096 систем, 128 прыжков) или слишком сложен.';
            07=if(status==-7)problem='Посадка на целевую планету запрещена.';
            08=MessageBox(problem);
        }
        Return ^{
            00=unknown prepare=ImportedFunction('FHDQuestNative','FHD_RoutePrepare');
            01=unknown summary=ImportedFunction('FHDQuestNative','FHD_RouteText');
            02=unknown cancel=ImportedFunction('FHDQuestNative','FHD_RouteCancel');
            03=int status=prepare();
            04=if(status)ExecuteCodeFromString(GenerateCodeStringFromBlock('Script.QuestMarker.Error'),'status',status);
            05=if(status)exit;
            06=unknown routeTarget=ImportedFunction('FHDQuestNative','FHD_RouteTarget');dword star=routeTarget();ExecuteCodeFromString(GenerateCodeStringFromBlock('Script.QuestMarker.Mark'),'star',star);
            07=InterfaceText('Galaxy','FHDQuestRouteInfo',summary());
            08=InterfaceState('Galaxy','FHDQuestRouteInfo',1);
            09=unknown deadline=ImportedFunction('FHDQuestNative','FHD_RouteDeadlineText');InterfaceText('Galaxy','FHDQuestRouteDeadline',deadline());InterfaceState('Galaxy','FHDQuestRouteDeadline',1);
            10=RunChildForm('Galaxy',2);
            11=cancel();
            12=InterfaceState('Galaxy','FHDQuestRouteInfo',-1);
            13=ShipDeleteCustomShipInfo(Player(),'FHDQuestMarkerState');InterfaceState('Galaxy','FHDQuestRouteDeadline',-1);InterfaceText('Galaxy','FHDQuestRouteDeadline','');InterfaceState('Galaxy','FHDQuestMarkV035',-1);
            14=InterfaceText('Galaxy','FHDQuestRouteInfo','');
        }
        Show ^{
            00=unknown getKind=ImportedFunction('FHDQuestNative','FHD_GetKind');
            01=unknown getTarget=ImportedFunction('FHDQuestNative','FHD_GetTarget');
            02=unknown reset=ImportedFunction('FHDQuestNative','FHD_Reset');
            03=int kind=getKind();
            04=if(kind<0)exit;
            05=dword target=getTarget();
            06=reset();
            07=if(!target)exit;
            08=dword star=0;
            09=if(kind==0 || kind==2)star=PlanetToStar(target);
            10=if(kind==1 || kind==4)star=ShipStar(target);
            11=if(kind==3)star=target;
            12=ExecuteCodeFromString(GenerateCodeStringFromBlock('Script.QuestMarker.Place'),'star',star);
        }
    }
}

ShipInfo ^{
    AddInfo ^{
        CustomInfos ^{
            FHDQuestMarkerState ^{
                Name=Служебная метка карты
                Description=Цель метки, применяемая после открытия галактической карты
                Icon=0
                OnActCodeTypes=t_OnEnteringForm
                OnActCode ~{
                    00=ExecuteCodeFromString(GenerateCodeStringFromBlock('Script.QuestMarker.ApplyPending'));
                }
            }
        }
    }
}
